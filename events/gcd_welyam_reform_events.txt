country_event = {
    id = gcd_welyam.90
    title = gcd_welyam.90.t
    desc = gcd_welyam_placeholder_desc
    picture = NOBLE_ESTATE_eventPicture
    hidden = yes
    is_triggered_only = yes
    trigger = {
        tag = A13
        OR = {
            has_disaster = gcd_welyam_reform
            has_country_flag = gcd_welyam_civil_war_active
        }
    }
    # 修正（12月12日栈溢出）：不能在 immediate 中安排自身。
    # 按 Anbennar diggy_dungeons.2000 的结构，在隐藏自动选项中执行并延迟下一轮。
    option = {
        name = gcd_welyam.90.a
        ai_chance = { factor = 1 }
        gcd_welyam_daily_reform_effect = yes
        # 新增：灾难/内战均结束后不再排队下一天，避免永久空转。
        if = {
            limit = {
                OR = {
                    has_disaster = gcd_welyam_reform
                    has_country_flag = gcd_welyam_civil_war_active
                }
            }
            country_event = { id = gcd_welyam.90 days = 1 }
        }
    }
}

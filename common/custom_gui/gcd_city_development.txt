# 城市发展署的可见性、点击与提示逻辑。
# 入口按钮和全屏锚点挂在经济界面 countryeconomyview 内。
# 仅通过“发展署”按钮切换窗口；首次点击同时初始化加韦顿。
# custom_gui 的当前作用域是玩家国家；先确认 ROOT 存在，再读取本国旗标。
custom_button = {
    name = gcd_gui_open
    # 入口位于经济页右上角，保留原有按钮样式。
    # 只有加韦德玩家能够点击；其他国家只会看到灰色入口。
    potential = { always = yes }
    trigger = { exists = yes tag = A13 ai = no }
    effect = {
        if = { limit = { has_country_flag = gcd_window_open } clr_country_flag = gcd_window_open }
        else = {
            if = {
                limit = { NOT = { has_country_flag = gcd_city_initialized } }
                set_country_flag = gcd_city_initialized
                gcd_gaweton_set_level_1 = yes
            }
            gcd_clear_city_selection = yes
            set_country_flag = gcd_gaweton_selected
            set_country_flag = gcd_window_open
        }
    }
    tooltip = gcd_entry_tooltip
}
custom_window = {
    name = gcd_city_anchor
    potential = { exists = ROOT exists = yes ai = no has_country_flag = gcd_window_open }
}
custom_window = {
    name = gcd_city_window
    potential = { exists = ROOT exists = yes ai = no }
}
custom_button = {
    name = gcd_close
    potential = { exists = ROOT exists = yes }
    trigger = { exists = yes }
    effect = { clr_country_flag = gcd_window_open }
    tooltip = gcd_close_tooltip
}
